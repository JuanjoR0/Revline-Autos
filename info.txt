from django.db import models
from django.contrib.auth.models import User

class Vehicle(models.Model):
  TYPE_CHOICES = [('car','Coche'),('bike','Moto'),('special','Especial')]
  name = models.CharField(max_length=120)
  brand = models.CharField(max_length=120)
  type = models.CharField(max_length=20, choices=TYPE_CHOICES)
  price = models.DecimalField(max_digits=10, decimal_places=2)
  stock = models.PositiveIntegerField(default=0)
  speed = models.DecimalField(max_digits=5, decimal_places=2)
  braking = models.DecimalField(max_digits=5, decimal_places=2)
  acceleration = models.DecimalField(max_digits=5, decimal_places=2)
  traction = models.DecimalField(max_digits=5, decimal_places=2)
  image = models.ImageField(upload_to="vehicles/", blank=True, null=True)
  created_at = models.DateTimeField(auto_now_add=True)

  def __str__(self): return self.name

class Order(models.Model):
  STATUS_CHOICES = [('PENDING','Pendiente'),('PAID','Pagado'),('CANCELLED','Cancelado')]
  user = models.ForeignKey(User, on_delete=models.CASCADE)
  address = models.CharField(max_length=255)
  postal_code = models.CharField(max_length=20)
  province = models.CharField(max_length=120)
  status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='PENDING')
  paid = models.BooleanField(default=False)
  created_at = models.DateTimeField(auto_now_add=True)

  def __str__(self): return f"Pedido #{self.id} - {self.user.username}"

class OrderItem(models.Model):
  order = models.ForeignKey(Order, related_name='items', on_delete=models.CASCADE)
  vehicle = models.ForeignKey(Vehicle, on_delete=models.PROTECT)
  unit_price = models.DecimalField(max_digits=10, decimal_places=2)
  quantity = models.PositiveIntegerField(default=1)

  def __str__(self): return f"{self.vehicle.name} x{self.quantity}"

# serializers.py
from rest_framework import serializers
from django.contrib.auth.models import User
from .models import Vehicle, Order, OrderItem

class RegisterSerializer(serializers.ModelSerializer):
  class Meta: model = User; fields = ['id','username','email','password']; extra_kwargs={'password':{'write_only':True}}
  def create(self, data): return User.objects.create_user(**data)

class VehicleSerializer(serializers.ModelSerializer):
  class Meta: model = Vehicle; fields = '__all__'

class OrderItemWriteSerializer(serializers.ModelSerializer):
  class Meta: model = OrderItem; fields = ['vehicle','unit_price','quantity']

class OrderItemReadSerializer(serializers.ModelSerializer):
  vehicle = VehicleSerializer()
  class Meta: model = OrderItem; fields = ['id','vehicle','unit_price','quantity']

class OrderSerializer(serializers.ModelSerializer):
  items = OrderItemReadSerializer(many=True, read_only=True)
  class Meta: model = Order; fields = ['id','user','address','postal_code','province','status','paid','created_at','items']
  read_only_fields = ['user','status','paid','created_at']

# views.py
from rest_framework import viewsets, generics, permissions, status
from rest_framework.response import Response
from .models import Vehicle, Order, OrderItem
from .serializers import VehicleSerializer, OrderSerializer, RegisterSerializer, OrderItemWriteSerializer

class VehicleViewSet(viewsets.ModelViewSet):
  queryset = Vehicle.objects.all().order_by('-created_at')
  serializer_class = VehicleSerializer
  permission_classes = [permissions.AllowAny]  # proteger POST/PUT/DELETE con IsAdminUser en producción
  def get_permissions(self):
    if self.action in ['create','update','partial_update','destroy']:
      return [permissions.IsAdminUser()]
    return super().get_permissions()

class OrderViewSet(viewsets.ModelViewSet):
  serializer_class = OrderSerializer
  permission_classes = [permissions.IsAuthenticated]
  def get_queryset(self):
    u = self.request.user
    return Order.objects.all() if u.is_staff else Order.objects.filter(user=u)
  def create(self, request, *args, **kwargs):
    order = Order.objects.create(
      user=request.user,
      address=request.data.get('address',''),
      postal_code=request.data.get('postal_code',''),
      province=request.data.get('province','')
    )
    items = request.data.get('items', [])
    for it in items:
      OrderItem.objects.create(order=order, **it)
    return Response(self.get_serializer(order).data, status=status.HTTP_201_CREATED)

class RegisterView(generics.CreateAPIView):
  serializer_class = RegisterSerializer
  permission_classes = [permissions.AllowAny]

12) DESPLIEGUE ÚNICO EN RENDER (DJANGO + REACT)
──────────────────────────────────────────────────────────────────────────────
1. Crea en Render:
   - Un servicio de Base de Datos PostgreSQL → copia DATABASE_URL.
   - Un Web Service desde tu repo (raíz del monorepo).
2. Variables de entorno (Environment):
   DATABASE_URL=postgres://...
   SECRET_KEY=<tu_clave_segura>
   DEBUG=False
   ALLOWED_HOSTS=.onrender.com
3. Build Command (en el Web Service):
   pip install -r requirements.txt && \
   npm --prefix frontend ci || npm --prefix frontend install && \
   npm --prefix frontend run build && \
   rm -rf backend/frontend_dist && cp -R frontend/dist backend/frontend_dist && \
   python backend/manage.py collectstatic --noinput && \
   python backend/manage.py migrate
4. Start Command:
   gunicorn backend.backend.wsgi:application
   # Nota: el módulo wsgi apunta a backend/backend/wsgi.py
5. Tras el primer despliegue, crea superusuario (Jobs o Shell):
   python backend/manage.py createsuperuser

Listo ✅ Una sola URL en Render servirá la SPA y la API.

NOTAS
──────────────────────────────────────────────────────────────────────────────
- En producción (mismo dominio) no necesitas CORS para la SPA.
- Si desarrollas el frontend con Vite en local, activa CORS para http://localhost:5173.
- Usa IsAdminUser para proteger POST/PUT/DELETE de vehículos.
- Añade paginación y búsqueda en VehicleViewSet si lo deseas (DRF LimitOffsetPagination / filtros).
- No subas db.sqlite3 al repo. Usa migraciones para crear tablas en Postgres.
